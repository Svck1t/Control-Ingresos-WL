Configurar sincronización

Conecta esta app a una hoja de Google para que tus datos se sincronicen entre dispositivos. Es un proceso de una sola vez por dispositivo.

1Crea una hoja de cálculo

Ve a sheets.google.com y crea una hoja nueva. Ponle el nombre que quieras, por ejemplo Control Compras.

2Abre Apps Script

Dentro de la hoja, ve al menú Extensiones → Apps Script. Se abre un editor de código en una nueva pestaña.

3Pega el código del servidor

Borra todo lo que aparece por defecto en el editor y pega este código completo:

const SHEET_NAME = 'Compras';
const HEADERS = ['id','fecha','proveedor','producto','cantidad','unidad','costoUnitario','estado','referencia','notas','fotoUrl','creadoEn','actualizadoEn'];

function getSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(HEADERS);
    sheet.getRange('A:B').setNumberFormat('@');
    sheet.getRange('H:J').setNumberFormat('@');
    sheet.setFrozenRows(1);
    sheet.getRange(1, 1, 1, HEADERS.length).setFontWeight('bold');
  }
  return sheet;
}

function doGet(e) {
  try {
    const sheet = getSheet();
    const data = sheet.getDataRange().getValues();
    if (data.length <= 1) return jsonOut({ ok: true, entries: [] });
    const headers = data[0];
    const entries = data.slice(1).map(row => {
      const obj = {};
      headers.forEach((h, i) => {
        let val = row[i];
        if (val instanceof Date) val = Utilities.formatDate(val, Session.getScriptTimeZone(), 'yyyy-MM-dd');
        obj[h] = val;
      });
      return obj;
    });
    return jsonOut({ ok: true, entries: entries });
  } catch (err) { return jsonOut({ ok: false, error: err.toString() }); }
}

function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents);
    const action = body.action;
    const sheet = getSheet();
    if (action === 'add') {
      if (body.image) {
        body.entry.fotoUrl = guardarFotoBoleta(body.image, body.mimeType, body.entry.id);
      }
      const row = HEADERS.map(h => body.entry[h] !== undefined ? body.entry[h] : '');
      sheet.appendRow(row);
      return jsonOut({ ok: true, entry: body.entry });
    }
    if (action === 'update') {
      const rowIdx = findRowById(sheet, body.entry.id);
      if (rowIdx === -1) return jsonOut({ ok: false, error: 'No encontrado' });
      if (body.image) {
        body.entry.fotoUrl = guardarFotoBoleta(body.image, body.mimeType, body.entry.id);
      }
      const row = HEADERS.map(h => body.entry[h] !== undefined ? body.entry[h] : '');
      sheet.getRange(rowIdx, 1, 1, HEADERS.length).setValues([row]);
      return jsonOut({ ok: true, entry: body.entry });
    }
    if (action === 'delete') {
      const rowIdx = findRowById(sheet, body.id);
      if (rowIdx === -1) return jsonOut({ ok: false, error: 'No encontrado' });
      sheet.deleteRow(rowIdx);
      return jsonOut({ ok: true });
    }
    if (action === 'bulk_add') {
      const rows = body.entries.map(entry => HEADERS.map(h => entry[h] !== undefined ? entry[h] : ''));
      if (rows.length > 0) sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, HEADERS.length).setValues(rows);
      return jsonOut({ ok: true, count: rows.length });
    }
    if (action === 'extract') {
      return jsonOut(extractFromImage(body.image, body.mimeType));
    }
    return jsonOut({ ok: false, error: 'Acción desconocida' });
  } catch (err) { return jsonOut({ ok: false, error: err.toString() }); }
}

function findRowById(sheet, id) {
  const data = sheet.getRange(2, 1, Math.max(sheet.getLastRow() - 1, 1), 1).getValues();
  for (let i = 0; i < data.length; i++) if (String(data[i][0]) === String(id)) return i + 2;
  return -1;
}

function jsonOut(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

// Devuelve (o crea la primera vez) la carpeta de Drive donde se guardan las fotos de boletas.
// El ID de la carpeta se guarda en las Propiedades del script para reutilizar siempre la misma.
function getFotosFolder() {
  const props = PropertiesService.getScriptProperties();
  const folderId = props.getProperty('FOTOS_FOLDER_ID');
  if (folderId) {
    try { return DriveApp.getFolderById(folderId); } catch (e) { /* la carpeta fue borrada, se crea otra abajo */ }
  }
  const folder = DriveApp.createFolder('Control de Compras - Fotos boletas');
  props.setProperty('FOTOS_FOLDER_ID', folder.getId());
  return folder;
}

// Guarda la imagen de una boleta en Drive y devuelve una URL para mostrarla en el navegador.
function guardarFotoBoleta(base64Image, mimeType, entryId) {
  const folder = getFotosFolder();
  const bytes = Utilities.base64Decode(base64Image);
  const blob = Utilities.newBlob(bytes, mimeType || 'image/jpeg', 'boleta_' + entryId + '.jpg');
  const file = folder.createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return 'https://lh3.googleusercontent.com/d/' + file.getId() + '=w1000';
}

// Extrae los datos de una boleta usando Gemini (Google AI).
// Requiere la propiedad de script GEMINI_API_KEY (Configuración del proyecto → Propiedades del script).
function extractFromImage(base64Image, mimeType) {
  const apiKey = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY');
  if (!apiKey) return { ok: false, error: 'Falta configurar GEMINI_API_KEY en las Propiedades del script.' };

  const prompt = 'Eres un asistente que extrae datos de boletas o facturas de compra en español (Chile). ' +
    'Analiza la imagen y devuelve SOLO un JSON valido (sin texto adicional, sin markdown) con esta forma exacta: ' +
    '{"proveedor": "...", "producto": "...", "cantidad": 0, "costoUnitario": 0}. ' +
    'Reglas: "proveedor" es el nombre del negocio que emite la boleta. "producto" es la descripcion del producto o servicio comprado ' +
    '(si hay varios items, usa el principal o resume). "cantidad" es un numero (usa 1 si no se especifica). ' +
    '"costoUnitario" es el precio unitario en pesos chilenos, solo el numero, sin puntos ni simbolos. ' +
    'No incluyas fecha, no es necesaria. Si no puedes leer algun campo, usa null en ese campo. No inventes datos que no esten en la imagen.';

  const payload = {
    contents: [{
      parts: [
        { text: prompt },
        { inline_data: { mime_type: mimeType || 'image/jpeg', data: base64Image } }
      ]
    }],
    generationConfig: { temperature: 0, response_mime_type: 'application/json' }
  };

  // Se usa el alias "flash-latest" para que no se rompa cuando Google retire una versión puntual
  // (esto ya pasó una vez con gemini-2.5-flash). Si igual deja de funcionar, revisa el modelo
  // vigente en la documentación de Gemini (ai.google.dev).
  const url = 'https://generativelanguage.googleapis.com/v1beta/models/gemini-flash-latest:generateContent?key=' + apiKey;
  const res = UrlFetchApp.fetch(url, {
    method: 'post',
    contentType: 'application/json',
    payload: JSON.stringify(payload),
    muteHttpExceptions: true,
  });

  const code = res.getResponseCode();
  const text = res.getContentText();
  if (code !== 200) return { ok: false, error: 'Error de Gemini (' + code + '): ' + text.slice(0, 300) };

  try {
    const json = JSON.parse(text);
    const raw = json.candidates[0].content.parts[0].text;
    const parsed = JSON.parse(raw);
    return { ok: true, data: parsed };
  } catch (err) {
    return { ok: false, error: 'No se pudo interpretar la respuesta de Gemini: ' + err };
  }
}

Guarda con Ctrl+S y ponle un nombre al proyecto (lo que quieras).

4Despliega como aplicación web

Arriba a la derecha, click en Implementar → Nueva implementación. En la ventana que aparece:

  • Click en el ícono de engranaje y selecciona Aplicación web
  • "Ejecutar como": Yo (tu correo)
  • "Quién tiene acceso": Cualquier usuario
  • Click Implementar

La primera vez te pedirá autorizar. Acepta los permisos. Si ves "Google no ha verificado esta app", click en Configuración avanzada → Ir a [nombre] (no seguro). Es tu propio script, está bien.

Al final aparece una URL larga que empieza con https://script.google.com/macros/s/... — cópiala.

5Pega la URL aquí
Seguridad: cualquiera con esa URL puede ver y modificar tu hoja. Guárdala como una contraseña. Si necesitas revocarla, crea una nueva implementación en Apps Script.
